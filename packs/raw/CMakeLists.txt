cmake_minimum_required(VERSION 3.25)
cmake_policy(SET CMP0091 NEW)
project(luling_rawpack LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
find_package(libraw CONFIG REQUIRED)
add_executable(luling-rawpack src/rawpack.cpp)
target_include_directories(luling-rawpack PRIVATE ../../src/common)
target_compile_definitions(luling-rawpack PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(luling-rawpack PRIVATE /W4 /EHsc /utf-8 /permissive-)
target_link_libraries(luling-rawpack PRIVATE libraw::raw_r windowscodecs ole32)
target_link_options(luling-rawpack PRIVATE /INCREMENTAL:NO /OPT:REF /OPT:ICF /DEPENDENTLOADFLAG:0xA00)
enable_testing()
add_test(NAME rawpack_formats COMMAND luling-rawpack formats)
